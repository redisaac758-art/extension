export const uid = (prefix = 'id') =>
  `${prefix}_${Math.random().toString(36).slice(2, 8)}${Date.now().toString(36).slice(-4)}`;

export const clamp = (n, lo, hi) => Math.min(hi, Math.max(lo, n));

export function abortError() {
  const e = new Error('Aborted');
  e.name = 'AbortError';
  return e;
}

export function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) return reject(abortError());
    const t = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();
    }, Math.max(0, ms));
    const onAbort = () => {
      clearTimeout(t);
      reject(abortError());
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

export function truncate(str, n = 200) {
  const s = String(str ?? '');
  return s.length > n ? s.slice(0, n - 1) + '…' : s;
}

export function hostOf(url) {
  try {
    return new URL(url).hostname.replace(/^www\./, '');
  } catch {
    return '';
  }
}

export function timeAgo(ts) {
  const s = Math.round((Date.now() - ts) / 1000);
  if (s < 45) return 'just now';
  if (s < 3600) return `${Math.round(s / 60)} min ago`;
  if (s < 86400) return `${Math.round(s / 3600)} h ago`;
  return `${Math.round(s / 86400)} d ago`;
}

/**
 * Pull the first JSON object out of a model reply. Handles ```json fences,
 * <think> blocks from reasoning models, prose around the object, and trailing commas.
 */
export function extractJson(text) {
  if (!text) return null;
  let s = String(text).replace(/<think>[\s\S]*?<\/think>/gi, '').trim();
  const fence = s.match(/```(?:json)?\s*([\s\S]*?)```/i);
  if (fence) s = fence[1].trim();
  const parse = (raw) => {
    try {
      return JSON.parse(raw);
    } catch {
      try {
        return JSON.parse(raw.replace(/,\s*([}\]])/g, '$1'));
      } catch {
        return null;
      }
    }
  };
  const direct = parse(s);
  if (direct && typeof direct === 'object') return direct;
  const start = s.indexOf('{');
  if (start < 0) return null;
  let depth = 0;
  let inStr = false;
  let esc = false;
  for (let i = start; i < s.length; i++) {
    const c = s[i];
    if (inStr) {
      if (esc) esc = false;
      else if (c === '\\') esc = true;
      else if (c === '"') inStr = false;
      continue;
    }
    if (c === '"') inStr = true;
    else if (c === '{') depth++;
    else if (c === '}') {
      depth--;
      if (depth === 0) return parse(s.slice(start, i + 1));
    }
  }
  return null;
}

/** "10.5" -> 0.05 (half a unit in the last typed decimal). Integers -> 0. */
export function toleranceFor(text) {
  const t = String(text).trim();
  const m = t.match(/\.(\d+)$/);
  if (!m) return 0;
  return 0.5 * Math.pow(10, -m[1].length);
}

export function downloadJson(filename, data) {
  const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.click();
  setTimeout(() => URL.revokeObjectURL(url), 2000);
}
