-- Game Agent: optional cloud sync.
-- Run this once in the Supabase SQL editor (Dashboard -> SQL Editor -> New query).
-- Every table is locked to its owner with row level security.
-- API keys are never uploaded by the extension.

create table if not exists public.ga_settings (
  user_id    uuid primary key references auth.users (id) on delete cascade,
  data       jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now()
);

create table if not exists public.ga_presets (
  user_id    uuid not null references auth.users (id) on delete cascade,
  id         text not null,
  data       jsonb not null,
  updated_at timestamptz not null default now(),
  primary key (user_id, id)
);

create table if not exists public.ga_models (
  user_id    uuid not null references auth.users (id) on delete cascade,
  id         text not null,
  data       jsonb not null,          -- model config without the api key
  updated_at timestamptz not null default now(),
  primary key (user_id, id)
);

create table if not exists public.ga_runs (
  user_id    uuid not null references auth.users (id) on delete cascade,
  id         text not null,
  data       jsonb not null,          -- goal, model, steps, result, transcript (no screenshots)
  created_at timestamptz not null default now(),
  primary key (user_id, id)
);

create index if not exists ga_runs_user_created on public.ga_runs (user_id, created_at desc);

alter table public.ga_settings enable row level security;
alter table public.ga_presets  enable row level security;
alter table public.ga_models   enable row level security;
alter table public.ga_runs     enable row level security;

do $$
declare
  t text;
begin
  foreach t in array array['ga_settings', 'ga_presets', 'ga_models', 'ga_runs'] loop
    execute format('drop policy if exists "owner full access" on public.%I', t);
    execute format(
      'create policy "owner full access" on public.%I for all to authenticated
         using (auth.uid() = user_id) with check (auth.uid() = user_id)', t);
  end loop;
end $$;
