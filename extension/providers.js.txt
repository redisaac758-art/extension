export const providerById = (id) => PROVIDERS.find((p) => p.id === id) || PROVIDERS[PROVIDERS.length - 1];

export const GROUP_LABELS = {
  free: 'Free hosted models',
  local: 'Run on your PC',
  paid: 'Bring your own key',
  custom: 'Custom',
};

/** Blank model config seeded from a provider. */
export function newModelFromProvider(providerId) {
  const p = providerById(providerId);
  const first = p.suggested[0];
  return {
    id: null,
    name: p.id === 'custom' ? 'My model' : p.name.replace(/ \(.*\)$/, ''),
    providerId: p.id,
    format: p.format,
    baseUrl: p.baseUrl,
    apiKey: '',
    model: first?.model || '',
    vision: first?.vision ?? false,
    rpm: p.rpm,
    temperature: 0.3,
    maxTokens: 900,
    headers: p.headers ? { ...p.headers } : {},
    extraBody: {},
    free: p.group === 'free' || p.group === 'local',
  };
}
